🔴 Vulnerable Code:

                            
// String concatenation - DANGEROUS!
const query = `SELECT * FROM users 
  WHERE username = '${searchTerm}'`;
db.exec(query);

                        

                        

                            
🟢 Secure Code:

                            
// Parameterized query - SAFE!
const stmt = db.prepare(
  'SELECT * FROM users WHERE username = ?'
);
stmt.bind([searchTerm]);
